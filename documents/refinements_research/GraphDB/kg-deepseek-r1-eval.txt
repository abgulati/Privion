# validation_service.py
from datetime import datetime
from typing import Dict, List

class GraphValidator:
    def __init__(self, node_schema: Dict, relationship_rules: Dict):
        self.node_schema = node_schema
        self.relationship_rules = relationship_rules
        self.date_format = "%Y-%m-%d"
        
    def validate_nodes(self, nodes: List[Dict]) -> bool:
        """Validate nodes against schema requirements"""
        checks = [
            self._check_mandatory_properties(node)
            and self._check_date_formats(node)
            and self._check_value_types(node)
            for node in nodes
        ]
        return all(checks)
    
    def validate_relationships(self, edges: Dict) -> bool:
        """Validate relationship consistency"""
        return all(
            self._check_relationship_validity(source_id, target_id, rel)
            for source_id, relationships in edges.items()
            for rel in relationships
            for target_id in [rel['id']]
        )

    def _check_mandatory_properties(self, node: Dict) -> bool:
        required = self.node_schema[node['type']]['required']
        return all(prop in node for prop in required)

    def _check_date_formats(self, node: Dict) -> bool:
        if 'date' in node:
            try:
                datetime.strptime(node['date'], self.date_format)
                return True
            except ValueError:
                return False
        return True

    def _check_value_types(self, node: Dict) -> bool:
        if node['type'] == 'FinancialResult':
            return isinstance(node.get('value'), (int, float))
        return True

    def _check_relationship_validity(self, source_id: str, target_id: str, rel: Dict) -> bool:
        rule = self.relationship_rules.get(rel['description'])
        if not rule:
            return False
            
        source_type = self._get_node_type(source_id)
        target_type = self._get_node_type(target_id)
        
        return (
            source_type in rule['allowed_source']
            and target_type in rule['allowed_target']
            and rel['confidence'] >= rule['min_confidence']
        )
